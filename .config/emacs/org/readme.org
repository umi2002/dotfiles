** Org-Agenda
Functions
#+begin_src emacs-lisp
  (defun org-habit-streak-count ()
    (goto-char (point-min))
    (while (not (eobp))
      (when (get-text-property (point) 'org-habit-p)
        (let ((pos (+ (line-beginning-position) org-habit-graph-column org-habit-preceding-days))
	      (streak 0))
	  (unless (eq (char-after pos) org-habit-completed-glyph)
	    (setq pos (1- pos)))
	  (while (eq (char-after pos) org-habit-completed-glyph)
	    (setq streak (+ streak 1))
	    (setq pos (1- pos)))
	  (end-of-line)
	  (insert (number-to-string streak))))
      (forward-line 1)))
#+end_src

Setup
#+begin_src emacs-lisp
  (setq org-agenda-files (list "inbox.org" "projects.org" "someday.org" "habits.org" "birthdays.org" "calendar.org"))

  (setq org-todo-keywords
        '((sequence "TODO(t)" "NEXT(n)" "WAITING(w@/!)" "|" "DONE(d)" "CANCELLED(c@)")))

  (setq org-log-done 'time)
  (setq org-log-into-drawer t)
  (setq org-enforce-todo-dependencies t)

  (setq org-refile-targets
        `((,(expand-file-name "projects.org" org-directory) :maxlevel . 2)
          (,(expand-file-name "someday.org" org-directory) :maxlevel . 1)
          (,(expand-file-name "habits.org" org-directory) :maxlevel . 1)))
  (add-hook 'org-after-refile-insert-hook #'save-buffer)
  (setq org-refile-use-outline-path 'file)
  (setq org-outline-path-complete-in-steps nil)

  (setq org-agenda-todo-ignore-scheduled 'future)
  (setq org-tags-exclude-from-inheritance '("noproject"))
  (setq org-stuck-projects '("+LEVEL=1-noproject/-DONE" ("NEXT" "WAITING") nil ""))

  (setq org-capture-templates
        `(("t" "Thought" entry
           (file ,(expand-file-name "inbox.org" org-directory))
           "* %?\n  %U")
          ("a" "Appointment" entry
           (file ,(expand-file-name "inbox.org" org-directory))
           "* %?\n  %^T")
          ("l" "Thought with context" entry
           (file ,(expand-file-name "inbox.org" org-directory))
           "* %?\n  %U\n  %a")))

  (setq org-agenda-custom-commands
        `(("g" "GTD"
           ((agenda ""
                    ((org-agenda-span 1)
                     (org-agenda-overriding-header "Today")))
            (tags "LEVEL=1"
                  ((org-agenda-files (list ,(expand-file-name "inbox.org" org-directory)))
                   (org-agenda-overriding-header "Inbox")))
            (todo "NEXT"
                  ((org-agenda-overriding-header "Next actions")))
            (todo "WAITING"
                  ((org-agenda-overriding-header "Waiting on")))
            (stuck ""
                   ((org-agenda-files (list ,(expand-file-name "projects.org" org-directory)))
                    (org-agenda-overriding-header "Stuck projects")))))))
#+end_src
